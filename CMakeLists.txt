function(game_object_library name)
  add_library(${name} OBJECT ${ARGN})
  target_compile_options(${name} PRIVATE ${GAME_C_FLAGS})
  target_include_directories(${name} PRIVATE ${GAME_INCLUDES})
endfunction()

# Quicksave snapshot bounds: these marker objects are the first and last of each executable, so the .data/.bss
# between them covers every port and game object and none of the system libraries wasm-ld links after them.
# CMake links $<TARGET_OBJECTS:...> before a target's own sources: qs_begin is an object library listed first,
# qs_end the executable's last own source.
foreach(edge begin end)
  file(CONFIGURE OUTPUT ${CMAKE_BINARY_DIR}/qs_markers/qs_${edge}.c CONTENT
"/* generated by CMakeLists.txt */
__attribute__((used, aligned(16))) char __qs_data_${edge}[16] = {1};
__attribute__((used, aligned(16))) char __qs_bss_${edge}[16];
")
endforeach()
add_library(dk64_qs_begin OBJECT ${CMAKE_BINARY_DIR}/qs_markers/qs_begin.c)

# Overlays: [begin marker, overlay TUs, overlay data, end marker] in link order. The markers bound the
# overlay's .data and .bss for portOverlayActivate (wasm clang ignores `#pragma clang section`).
foreach(ovl ${OVERLAYS})
  foreach(edge begin end)
    set(marker ${CMAKE_BINARY_DIR}/ovl_markers/ovl_${ovl}_${edge}.c)
    file(CONFIGURE OUTPUT ${marker} CONTENT
"/* generated by CMakeLists.txt */
__attribute__((used, aligned(16))) char __dk_${ovl}_data_${edge}[16] = {1};
__attribute__((used, aligned(16))) char __dk_${ovl}_bss_${edge}[16];
")
  endforeach()
  file(GLOB ovl_src CONFIGURE_DEPENDS ${TREE}/src/${ovl}/*.c)
  list(SORT ovl_src)
  set_source_files_properties(${ovl_src} PROPERTIES
    COMPILE_OPTIONS "-include;${GEN}/include/ovl/${ovl}.h")
  game_object_library(dk64_ovl_${ovl}
    ${CMAKE_BINARY_DIR}/ovl_markers/ovl_${ovl}_begin.c ${ovl_src} ${GEN}/data_${ovl}.c
    ${CMAKE_BINARY_DIR}/ovl_markers/ovl_${ovl}_end.c)
  list(APPEND GAME_OBJECTS $<TARGET_OBJECTS:dk64_ovl_${ovl}>)
  list(APPEND GAME_TARGETS dk64_ovl_${ovl})
endforeach()

# RSP task HLE and the WebGL backend never reach a fiber switch: keep asyncify's instrumentation out of them
# (tools/asyncify_remove.py lists every function their objects define; ASSERTIONS makes asyncify trap if a listed
# function is ever on a switching stack). The headless build uses it too, so its long regression runs exercise the
# list under those asserts.
find_program(LLVM_NM llvm-nm HINTS ${EMSCRIPTEN_ROOT_PATH}/../bin NO_DEFAULT_PATH REQUIRED)
function(asyncify_remove_list target)
  set(objs)
  foreach(lib ${ARGN})
    list(APPEND objs $<TARGET_OBJECTS:${lib}>)
  endforeach()
  set(out ${CMAKE_BINARY_DIR}/${target}_asyncify_remove.txt)
  add_custom_command(OUTPUT ${out}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/asyncify_remove.py ${LLVM_NM} ${out} ${objs}
    DEPENDS ${CMAKE_SOURCE_DIR}/tools/asyncify_remove.py ${ARGN} ${objs} VERBATIM COMMAND_EXPAND_LISTS)
  add_custom_target(${target}_asyncify_remove DEPENDS ${out})
  add_dependencies(${target} ${target}_asyncify_remove)
  target_link_options(${target} PRIVATE -sASYNCIFY_REMOVE=@${out})
  set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS ${out})
endfunction()

# The page (web/) is built by Vite -- typecheck, then bundle -- into the build dir next to dk64.js / dk64.wasm. It is
# its own target, so a JS-only edit rebuilds the page without relinking dk64. The source tree is globbed
# (CONFIGURE_DEPENDS re-runs the glob at build time) so a new module or asset needs no edit here; dotfiles and unit
# tests never reach the page. npm ci re-runs when package.json / package-lock.json change. OFF builds only the wasm
# (the Docker image builds the page in its own Node stage).
option(WEBBY_KONG_WEB "Build the page (web/) with npm + Vite into the build directory" ON)
if(WEBBY_KONG_WEB)
  find_program(NPM_EXECUTABLE npm REQUIRED)
  set(NPM_STAMP ${CMAKE_SOURCE_DIR}/node_modules/.package-lock.json)
  add_custom_command(OUTPUT ${NPM_STAMP}
    COMMAND ${NPM_EXECUTABLE} ci --no-audit --no-fund
    DEPENDS ${CMAKE_SOURCE_DIR}/package.json ${CMAKE_SOURCE_DIR}/package-lock.json
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "npm ci"
    VERBATIM)
  file(GLOB_RECURSE WEB_SRC CONFIGURE_DEPENDS LIST_DIRECTORIES false ${CMAKE_SOURCE_DIR}/web/*)
  list(FILTER WEB_SRC EXCLUDE REGEX "/web/(.*/)?[.]|[.]test[.]tsx?$")
  set(WEB_OUT ${CMAKE_BINARY_DIR}/index.html ${CMAKE_BINARY_DIR}/auth/index.html ${CMAKE_BINARY_DIR}/dashboard/index.html)
  add_custom_command(OUTPUT ${WEB_OUT}
    COMMAND ${NPM_EXECUTABLE} run typecheck
    COMMAND ${NPM_EXECUTABLE} exec -- vite build --outDir ${CMAKE_BINARY_DIR}
    DEPENDS ${WEB_SRC} ${NPM_STAMP} ${CMAKE_SOURCE_DIR}/vite.config.ts ${CMAKE_SOURCE_DIR}/tsconfig.json
    WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}
    COMMENT "Building the page (tsc + vite)"
    VERBATIM)
  add_custom_target(dk64_web_files ALL DEPENDS ${WEB_OUT})
endif()
