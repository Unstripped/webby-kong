# ---------------------------------------------------------------- game TUs
set(GAME_C_FLAGS
  -O2 -g -std=gnu89 -fsigned-char -fno-strict-aliasing -fwrapv -ffp-contract=off -fno-builtin -mnontrapping-fptoint
  ${COMMON_DEFS}
  "SHELL:-include ${CMAKE_SOURCE_DIR}/port/include/port/prelude.h"
  -Wno-unknown-pragmas -Wno-int-conversion -Wno-incompatible-pointer-types -Wno-deprecated-non-prototype -Wno-pointer-sign
  -Werror=implicit-function-declaration -Werror=return-type)
set(GAME_INCLUDES
  ${CMAKE_SOURCE_DIR}/port/include/override ${CMAKE_SOURCE_DIR}/port/include ${GEN}/include
  ${TREE}/include ${TREE}/include/PR ${TREE}/include/libultra ${TREE}/include/compiler/ido)

file(GLOB GAME_GLOBAL_SRC CONFIGURE_DEPENDS ${TREE}/src/global/*.c)
file(GLOB GAME_GU_SRC CONFIGURE_DEPENDS ${TREE}/src/boot/libultra/gu/*.c ${TREE}/src/boot/libultra/mgu/*.c)
list(SORT GAME_GLOBAL_SRC)
list(SORT GAME_GU_SRC)
set(GAME_BOOT_SRC
  ${TREE}/src/boot/1CC0.c ${TREE}/src/boot/30E0.c ${TREE}/src/boot/3630.c
  ${TREE}/src/boot/3D20.c ${TREE}/src/boot/6690.c
  ${TREE}/src/boot/libultra/libc/ll.c ${TREE}/src/boot/libultra/libc/llcvt.c)

function(game_object_library name)
  add_library(${name} OBJECT ${ARGN})
  target_compile_options(${name} PRIVATE ${GAME_C_FLAGS})
  target_include_directories(${name} PRIVATE ${GAME_INCLUDES})
endfunction()

game_object_library(dk64_game_global ${GAME_BOOT_SRC} ${GAME_GU_SRC} ${GAME_GLOBAL_SRC} ${GEN}/data_boot.c ${GEN}/data_global.c)
set(GAME_OBJECTS $<TARGET_OBJECTS:dk64_game_global>)
set(GAME_TARGETS dk64_game_global)

# Overlays: [begin marker, overlay TUs, overlay data, end marker] in link order. The markers bound the
# overlay's .data and .bss for portOverlayActivate (wasm clang ignores `#pragma clang section`).
foreach(ovl ${OVERLAYS})
  foreach(edge begin end)
    set(marker ${CMAKE_BINARY_DIR}/ovl_markers/ovl_${ovl}_${edge}.c)
    file(CONFIGURE OUTPUT ${marker} CONTENT
"/* generated by CMakeLists.txt */
__attribute__((used, aligned(16))) char __dk_${ovl}_data_${edge}[16] = {1};
__attribute__((used, aligned(16))) char __dk_${ovl}_bss_${edge}[16];
")
  endforeach()
  file(GLOB ovl_src CONFIGURE_DEPENDS ${TREE}/src/${ovl}/*.c)
  list(SORT ovl_src)
  set_source_files_properties(${ovl_src} PROPERTIES
    COMPILE_OPTIONS "-include;${GEN}/include/ovl/${ovl}.h")
  game_object_library(dk64_ovl_${ovl}
    ${CMAKE_BINARY_DIR}/ovl_markers/ovl_${ovl}_begin.c ${ovl_src} ${GEN}/data_${ovl}.c
    ${CMAKE_BINARY_DIR}/ovl_markers/ovl_${ovl}_end.c)
  list(APPEND GAME_OBJECTS $<TARGET_OBJECTS:dk64_ovl_${ovl}>)
  list(APPEND GAME_TARGETS dk64_ovl_${ovl})
endforeach()

add_custom_target(dk64_game DEPENDS ${GAME_TARGETS})

# CHECK_NULL_WRITES=0: wasm address 0 is RDRAM 0x80000000, which the game legitimately owns.
set(LINK_COMMON
  -sASYNCIFY=1 -sASYNCIFY_STACK_SIZE=65536 -sEMULATE_FUNCTION_POINTER_CASTS=1 -sGLOBAL_BASE=8MB
  -sINITIAL_MEMORY=64MB -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=512MB -sSTACK_SIZE=1MB -sCHECK_NULL_WRITES=0
  -Wl,--error-limit=0)

# fail the link on wasm-ld function signature mismatches
set(LINK_CHECK ${CMAKE_SOURCE_DIR}/tools/link_check.sh)

# ---------------------------------------------------------------- headless (node)

target_compile_options(dk64_headless PRIVATE ${PORT_C_FLAGS})

  set(HEADLESS_DEBUG_FLAGS -sSAFE_HEAP=2)

  -sEXPORTED_FUNCTIONS=_main,_port_debug_stats -sASSERTIONS=1 -Wl,-Map=${CMAKE_BINARY_DIR}/dk64_headless.map
  ${HEADLESS_DEBUG_FLAGS})
set_target_properties(dk64_headless PROPERTIES SUFFIX ".js" RULE_LAUNCH_LINK ${LINK_CHECK})
